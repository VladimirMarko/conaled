"""Finite formula checks for the proposed mod-four dependent fiber.
These are Python checks of paper formulas, not Lean interface admission.
"""
from itertools import product
Q = range(2)
R = range(4)

def fiber(m, n, r):
    return m == (n + r % 2) % 2

def total(m, n, a, r):
    return m == (n + a + r % 2) % 2

f = [(m, n, r) for m, n, r in product(Q, Q, R) if fiber(m, n, r)]
d = [(m, n, a, r) for m, n, a, r in product(Q, Q, Q, R) if total(m, n, a, r)]
for m, n, r in f:
    assert fiber(n, m, -r % 4)
    for k in Q:
        assert fiber((m+k)%2, (n+k)%2, r)
    for p, q, s in f:
        assert fiber((m+p)%2, (n+q)%2, (r+s)%4)
        if n == p:
            assert fiber(m, q, (r+s)%4)
for m, n, a, r in d:
    assert total(n, m, -a%2, -r%4)
    for k in Q:
        assert total((m+k)%2, (n+k)%2, a, r)
    for p, q, b, s in d:
        assert total((m+p)%2, (n+q)%2, (a+b)%2, (r+s)%4)
        if n == p:
            assert total(m, q, (a+b)%2, (r+s)%4)
for a, b, c, e in product(R, repeat=4):
    assert ((a+b)+(c+e))%4 == ((a+c)+(b+e))%4
for a, b, c in product(R, repeat=3):
    assert ((a+b)+c)%4 == (a+(b+c))%4
assert total(0, 0, 1, 1) and total(0, 0, 1, 3)
assert not total(0, 0, 1, 0)
witness_one, witness_three = (0, 0, 1, 1), (0, 0, 1, 3)
assert witness_one[:3] == witness_three[:3] and witness_one[3] != witness_three[3]
assert total(0, 0, 0, 2) and (0, 2) != (0, 0)
assert ((1+1)%2, (1+3)%4) == (0, 0)
for k in Q:
    assert total(k, k, 1, 1) and total(k, k, 1, 3)
# Independent Boolean encoding preserves the group operations used on Q.
for a, b in product([False, True], repeat=2):
    assert int(a ^ b) == (int(a)+int(b))%2
print(f'PASS: {len(f)} fiber cells, {len(d)} total cells; all inverse, vertical, horizontal, and shift boundary checks.')
print('PASS: mod-four associativity/interchange; parallel witnesses 1 and 3; zero replacement rejected; Boolean/mod-two operation preservation.')
print('Not checked: Lean Bicategory/TwoFamily admission, dependent transports, comma functors, or their universal laws.')
