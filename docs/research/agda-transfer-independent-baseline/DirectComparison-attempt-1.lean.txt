universe u v w

theorem directAssoc {C : Type u} [Category.{v} C]
    {x y z t : C} (p : x ⟶ y) (q : y ⟶ z) (r : z ⟶ t) :
    (p ≫ q) ≫ r = p ≫ (q ≫ r) := Category.assoc p q r

theorem directAction {C : Type u} [Category.{v} C] (B : C ⥤ Type w)
    {x y z t : C} (p : x ⟶ y) (q : y ⟶ z) (r : z ⟶ t)
    (b : B.obj x) :
    B.map ((p ≫ q) ≫ r) b = B.map (p ≫ (q ≫ r)) b :=
  congrArg (fun s => B.map s b) (directAssoc p q r)

open Examples.GroupoidTransport

theorem c4Assoc (p q r : point ⟶ point) :
    (p ≫ q) ≫ r = p ≫ (q ≫ r) := directAssoc p q r

theorem faithfulAction (p q r : point ⟶ point) (b : ZMod 4) :
    faithful.map ((p ≫ q) ≫ r) b = faithful.map (p ≫ (q ≫ r)) b :=
  directAction faithful p q r b

theorem trivialAction (p q r : point ⟶ point) (b : Fin 2) :
    ineffective.map ((p ≫ q) ≫ r) b = ineffective.map (p ≫ (q ≫ r)) b :=
  directAction ineffective p q r b

theorem labeledComposite : ((label 1 ≫ label 1) ≫ label 1) = label 3 := by decide

theorem faithfulValues :
    faithful.map ((label 1 ≫ label 1) ≫ label 1) (0 : ZMod 4) = 3 ∧
    faithful.map (label 1 ≫ (label 1 ≫ label 1)) (0 : ZMod 4) = 3 := by
  constructor
  · decide
  · rw [← faithfulAction]; decide

theorem trivialValues :
    ineffective.map ((label 1 ≫ label 1) ≫ label 1) (0 : Fin 2) = 0 ∧
    ineffective.map (label 1 ≫ (label 1 ≫ label 1)) (0 : Fin 2) = 0 := by
  constructor
  · rfl
  · rw [← trivialAction]; rfl
