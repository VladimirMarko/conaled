15: # The `Fintype` derive handler

16: 

17: This file defines a derive handler to automatically generate `Fintype` instances

18: for structures and inductive types.

19: 

20: The following is a prototypical example of what this can handle:

21: ```

22: inductive MyOption (α : Type*)

23:   | none

24:   | some (x : α)

25:   deriving Fintype

26: ```

27: 

28: This deriving handler does not attempt to process inductive types that are either

29: recursive or that have indices.

30: 

31: To get debugging information, do `set_option trace.Elab.Deriving.fintype true`

32: and `set_option Elab.ProxyType true`.

33: 

34: There is a term elaborator `derive_fintype%` implementing the derivation of `Fintype` instances.

35: This can be useful in cases when there are necessary additional assumptions (like `DecidableEq`).

36: This is implemented using `Fintype.ofEquiv` and `proxy_equiv%`, which is a term elaborator

37: that creates an equivalence from a "proxy type" composed of basic type constructors. If Lean

38: can synthesize a `Fintype` instance for the proxy type, then `derive_fintype%` succeeds.

39: 

40: ## Implementation notes

41: 

42: There are two kinds of `Fintype` instances that we generate, depending on the inductive type.

43: 

44: If it is an enum (an inductive type with only 0-ary constructors), then we generate the

45: complete `List` of all constructors; see `Mathlib.Deriving.Fintype.mkFintypeEnum` for more

46: details. The proof has $O(n)$ complexity in the number of constructors.

47: 

48: Otherwise, the strategy we take is to generate a "proxy type", define an equivalence between

49: our type and the proxy type (see `proxy_equiv%`), and then use `Fintype.ofEquiv` to pull a

50: `Fintype` instance on the proxy type (if one exists) to our inductive type. For example, with

51: the `MyOption α` type above, we generate `Unit ⊕ α`. While the proxy type is not a finite type

52: in general, we add `Fintype` instances for every type parameter of our inductive type (and

53: `Decidable` instances for every `Prop` parameter). Hence, in our example we get

54: `Fintype (MyOption α)` assuming `Fintype α`.

55: 

56: There is a source of quadratic complexity in this `Fintype` instance from the fact that an

57: inductive type with `n` constructors has a proxy type of the form `C₁ ⊕ (C₂ ⊕ (⋯ ⊕ Cₙ))`,

58: so mapping to and from `Cᵢ` requires looking through `i` levels of `Sum` constructors.

59: Ignoring time spent looking through these constructors, the construction of `Finset.univ`

60: contributes just linear time with respect to the cardinality of the type since the instances

61: involved compute the underlying `List` for the `Finset` as `l₁ ++ (l₂ ++ (⋯ ++ lₙ))` with

62: right associativity.

63: 

64: Note that an alternative design could be that instead of using `Sum` we could create a

65: function `C : Fin n → Type*` with `C i = ULift Cᵢ` and then use `(i : Fin n) × C i` for

66: the proxy type, which would save us from the nested `Sum` constructors.

67: 

68: This implementation takes some inspiration from the one by Mario Carneiro for Mathlib 3.

69: A difference is that the Mathlib 3 version does not explicitly construct the total proxy type,

70: and instead it opts to construct the underlying `Finset` as a disjoint union of the `Finset.univ`

71: for each individual constructor's proxy type.

80: /--

81: The term elaborator `derive_fintype% α` tries to synthesize a `Fintype α` instance

82: using all the assumptions in the local context; this can be useful, for example, if one

83: needs an extra `DecidableEq` instance. It works only if `α` is an inductive

84: type that `proxy_equiv% α` can handle. The elaborator makes use of the

85: expected type, so `(derive_fintype% _ : Fintype α)` works.

86: 

87: This uses `proxy_equiv% α`, so as a side effect it defines `proxyType` and `proxyTypeEquiv` in

88: the namespace associated to the inductive type `α`.

89: -/

90: macro "derive_fintype% " t:term : term => `(term| Fintype.ofEquiv _ (proxy_equiv% $t))

91: 

92: /-

93: Creates a `Fintype` instance by adding additional `Fintype` and `Decidable` instance arguments

94: for every type and prop parameter of the type, then use the `derive_fintype%` elaborator.

95: -/

96: def mkFintype (declName : Name) : CommandElabM Bool := do

97:   let indVal ← getConstInfoInduct declName

98:   let cmd ← liftTermElabM do

99:     let header ← Deriving.mkHeader `Fintype 0 indVal

100:     let binders' ← Deriving.mkInstImplicitBinders `Decidable indVal header.argNames

101:     let instCmd ← `(command|

102:       instance $header.binders:bracketedBinder* $(binders'.map TSyntax.mk):bracketedBinder* :

103:           Fintype $header.targetType := derive_fintype% _)

104:     return instCmd

105:   trace[Elab.Deriving.fintype] "instance command:\n{cmd}"

106:   elabCommand cmd

107:   return true

108: 

109: /-- Derive a `Fintype` instance for enum types. These come with a `ctorIdx` function.

110: 

111: We generate a more optimized instance than the one produced by `mkFintype`.

112: The strategy is to (1) create a list `enumList` of all the constructors, (2) prove that this

113: is in `ctorIdx` order, (3) show that `ctorIdx` maps `enumList` to `List.range numCtors` to show

114: the list has no duplicates, and (4) give the `Fintype` instance, using 2 for completeness.

115: 

116: The proofs are all linear complexity, and the main computation is that

117: `enumList.map ctorIdx = List.range numCtors`, which is true by `refl`. -/

118: def mkFintypeEnum (declName : Name) : CommandElabM Unit := do

119:   let indVal ← getConstInfoInduct declName

120:   let levels := indVal.levelParams.map Level.param

121:   let ctorIdxName := declName.mkStr "ctorIdx"

122:   let enumListName := declName.mkStr "enumList"

123:   let ctorThmName := declName.mkStr "enumList_getElem?_ctorIdx_eq"

124:   let enumListNodupName := declName.mkStr "enumList_nodup"

125:   liftTermElabM <| Term.withoutErrToSorry do

126:     do -- Define `enumList` enumerating all constructors

127:       trace[Elab.Deriving.fintype] "defining {enumListName}"

128:       let type := mkConst declName levels

129:       let listType ← mkAppM ``List #[type]

130:       let listNil ← mkAppOptM ``List.nil #[some type]

131:       let listCons name xs := mkAppM ``List.cons #[mkConst name levels, xs]

132:       let enumList ← indVal.ctors.foldrM (listCons · ·) listNil

133:       addAndCompile <| Declaration.defnDecl

134:         { name := enumListName

135:           levelParams := indVal.levelParams

136:           safety := DefinitionSafety.safe

137:           hints := ReducibilityHints.abbrev

138:           type := listType

139:           value := enumList }

140:       setProtected enumListName

141:       addDocStringCore enumListName s!"A list enumerating every element of the type, \

142:         which are all zero-argument constructors. (Generated by the `Fintype` deriving handler.)"

143:     do -- Prove that this list is in `ctorIdx` order

144:       trace[Elab.Deriving.fintype] "proving {ctorThmName}"

145:       let goalStx ← `(term| ∀ (x : $(← Term.exprToSyntax <| mkConst declName levels)),

146:         $(mkIdent enumListName)[$(mkIdent ctorIdxName) x]? = some x)

147:       let goal ← Term.elabTerm goalStx (mkSort .zero)

148:       let pf ← Term.elabTerm (← `(term| by intro x; cases x <;> rfl)) goal

149:       Term.synthesizeSyntheticMVarsNoPostponing

150:       addAndCompile <| Declaration.thmDecl

151:         { name := ctorThmName

152:           levelParams := indVal.levelParams

153:           type := ← instantiateMVars goal

154:           value := ← instantiateMVars pf }

155:       setProtected ctorThmName

156:     do -- Use this theorem to prove `enumList` has no duplicates

157:       trace[Elab.Deriving.fintype] "proving {enumListNodupName}"

158:       let enum ← Term.exprToSyntax <| mkConst enumListName levels

159:       let goal ← Term.elabTerm (← `(term| List.Nodup $enum)) (mkSort .zero)

160:       let n : TSyntax `term := quote indVal.numCtors

161:       let pf ← Term.elabTerm (← `(term| by

162:                   apply List.Nodup.of_map $(mkIdent ctorIdxName)

163:                   have h : List.map $(mkIdent ctorIdxName) $(mkIdent enumListName)

164:                             = List.range $n := rfl

165:                   exact h ▸ List.nodup_range)) goal

166:       Term.synthesizeSyntheticMVarsNoPostponing

167:       addAndCompile <| Declaration.thmDecl

168:         { name := enumListNodupName

169:           levelParams := indVal.levelParams

170:           type := ← instantiateMVars goal

171:           value := ← instantiateMVars pf }

172:       setProtected enumListNodupName

173:   -- Make the Fintype instance

174:   trace[Elab.Deriving.fintype] "defining fintype instance"

175:   let cmd ← `(command|

176:     instance : Fintype $(mkIdent declName) where

177:       elems := Finset.mk $(mkIdent enumListName) $(mkIdent enumListNodupName)

178:       complete := by

179:         intro x

180:         rw [Finset.mem_mk, Multiset.mem_coe, List.mem_iff_getElem?]

181:         exact ⟨$(mkIdent ctorIdxName) x, $(mkIdent ctorThmName) x⟩)

182:   trace[Elab.Deriving.fintype] "instance command:\n{cmd}"

183:   elabCommand cmd

184: 

185: def mkFintypeInstanceHandler (declNames : Array Name) : CommandElabM Bool := do

186:   if h : declNames.size ≠ 1 then

187:     return false -- mutually inductive types are not supported

188:   else

189:     let declName := declNames[0]

190:     if ← isEnumType declName then

191:       mkFintypeEnum declName

192:       return true

193:     else

194:       mkFintype declName

195: 

196: initialize

197:   registerDerivingHandler ``Fintype mkFintypeInstanceHandler

198:   registerTraceClass `Elab.Deriving.fintype

199: 

200: end Mathlib.Deriving.Fintype
