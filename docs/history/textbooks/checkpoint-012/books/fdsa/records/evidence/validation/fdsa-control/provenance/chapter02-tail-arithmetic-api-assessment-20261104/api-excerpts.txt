258: When `noIndexAtArgs := true`, `pushArgs` assumes function application arguments have a `no_index` annotation.
259: That is, `f a b` is indexed as it was `f (no_index a) (no_index b)`.
260: This feature is used when indexing local proofs in the simplifier. This is useful in examples like the one described on issue #2670.
261: In this issue, we have a local hypotheses `(h : ∀ p : α × β, f p p.2 = p.2)`, and users expect it to be applicable to
262: `f (a, b) b = b`. This worked in Lean 3 since no indexing was used. We can retrieve Lean 3 behavior by writing
263: `(h : ∀ p : α × β, f p (no_index p.2) = p.2)`, but this is very inconvenient when the hypotheses was not written by the user in first place.
264: For example, it was introduced by another tactic. Thus, when populating the discrimination tree explicit arguments provided to `simp` (e.g., `simp [h]`),
265: we use `noIndexAtArgs := true`. See comment: https://github.com/leanprover/lean4/issues/2670#issuecomment-1758889365
266: -/
267: private def pushArgs (root : Bool) (todo : Array Expr) (e : Expr) (noIndexAtArgs : Bool) : MetaM (Key × Array Expr) := do
268:   if hasNoindexAnnotation e then
269:     return (.star, todo)
270:   else
271:     let e ← reduceDT e root
272:     let fn := e.getAppFn
273:     let push (k : Key) (nargs : Nat) (todo : Array Expr): MetaM (Key × Array Expr) := do
274:       let info ← getFunInfoNArgs fn nargs
275:       let todo ← if noIndexAtArgs then
276:         pure <| pushWildcards nargs todo
277:       else
278:         pushArgsAux info.paramInfo (nargs-1) e todo
279:       return (k, todo)
280:     match fn with
281:     | .lit v     =>
282:       return (.lit v, todo)
283:     | .const c _ =>
284:       unless root do
285:         if let some v := toNatLit? e then
286:           return (.lit v, todo)
287:         if (← shouldAddAsStar c e) then
288:           return (.star, todo)
289:       let nargs := e.getAppNumArgs
290:       push (.const c nargs) nargs todo
