def level : BinaryTree α → Nat → List α
  | .nil, _ => []
  | .node a _ _, 0 => [a]
  | .node _ l r, k + 1 => level l k ++ level r k

@[simp] theorem level_nil (k : Nat) : level (.nil : BinaryTree α) k = []

@[simp] theorem level_nil (k : Nat) : level (.nil : BinaryTree α) k = []

@[simp] theorem level_node_zero (a : α) (l r : BinaryTree α) :
    level (.node a l r) 0 = [a]

@[simp] theorem level_node_succ (a : α) (l r : BinaryTree α) (k : Nat) :
    level (.node a l r) (k + 1) = level l k ++ level r k

def mergeLevels : List (List α) → List (List α) → List (List α)
  | [], ys => ys
  | xs, [] => xs
  | x :: xs, y :: ys => (x ++ y) :: mergeLevels xs ys

/-- Collect all levels by one structural traversal of the input tree. -/
def levels : BinaryTree α → List (List α)
  | .nil => []
  | .node a l r => [a] :: mergeLevels (levels l) (levels r)

@[simp] theorem levels_nil : levels (.nil : BinaryTree α) = []

theorem mergeLevels_length (xs ys : List (List α)) :
    (mergeLevels xs ys).length = max xs.length ys.length

theorem mergeLevels_getD (xs ys : List (List α)) (k : Nat) :
    (mergeLevels xs ys).getD k [] = xs.getD k [] ++ ys.getD k []

theorem levels_length (t : BinaryTree α) : (levels t).length = t.height

theorem levels_getD (t : BinaryTree α) (k : Nat) :
    (levels t).getD k [] = level t k

theorem levels_getElem? (t : BinaryTree α) (k : Nat) (hk : k < t.height) :
    (levels t)[k]? = some (level t k)

## Applications/Trees/Rose.lean

inductive Rose (α : Type u)

def mirror (t : Rose α) : Rose α

@[simp] theorem mirror_node (a : α) (children : List (Rose α)) :
    mirror (.node a children) = .node a ((children.map mirror).reverse)

theorem mirror_mirror (t : Rose α) : mirror (mirror t) = t
