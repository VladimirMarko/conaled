import Examples.ConversionTrial.Initial
set_option autoImplicit false
set_option warningAsError true
namespace Examples.ConversionTrial.Initial
open SquareKit.Internal.DependentFunctions

theorem source_derivation : DepEq lhs rhs := by
  have h : DepEq ((liftRen 1).depMap (source 1))
      ((liftRen 1).depMap target) :=
    DepEq.app (FuncEq.eta (.var (.base .top))) (.neg (.var 1))
  exact DepEq.sub (liftedNegate 1) (liftedNegate_typed 1)
    (DepEq.sub (liftedNegate 1) (liftedNegate_typed 1) h)

theorem lhs_index : lhs.type = index := rfl
theorem rhs_index : rhs.type = index := rfl

theorem task_proof : Task := ⟨source_derivation, lhs_index, rhs_index⟩
end Examples.ConversionTrial.Initial
