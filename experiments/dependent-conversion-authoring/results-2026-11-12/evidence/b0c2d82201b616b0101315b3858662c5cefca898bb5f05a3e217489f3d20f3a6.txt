import SquareKit.Internal.DependentFunctionTyping

/-! Experimental beta/eta equality generated by source rules. Substitution is
an operation on finite expression trees, including under the base binder. -/
namespace SquareKit.Internal.DependentFunctions
open DependentStructural (Base)

mutual
inductive DepEq : {n : Nat} → {Γ : Ctx n} → Dep Γ → Dep Γ → Prop where
  | refl {n} {Γ : Ctx n} (t : Dep Γ) : DepEq t t
  | symm : DepEq t u → DepEq u t
  | trans : DepEq t u → DepEq u v → DepEq t v
  | app {n} {Γ : Ctx n} {f g : Func Γ} (h : FuncEq f g) (a : Base n) :
      DepEq (.app f a) (.app g a)
  | beta {n} {Γ : Ctx n} (t : Dep (.base Γ)) (a : Base n) :
      DepEq (.app (.lam t) a) ((Sub.plug Γ a).depSub t)
  | sub {n m} {Γ : Ctx n} {Δ : Ctx m} (σ : Sub Δ Γ) (hs : σ.Typed)
      {t u : Dep Γ} : DepEq t u → DepEq (σ.depSub t) (σ.depSub u)
inductive FuncEq : {n : Nat} → {Γ : Ctx n} → Func Γ → Func Γ → Prop where
  | refl {n} {Γ : Ctx n} (f : Func Γ) : FuncEq f f
  | symm : FuncEq f g → FuncEq g f
  | trans : FuncEq f g → FuncEq g h → FuncEq f h
  | lam {n} {Γ : Ctx n} {t u : Dep (.base Γ)} : DepEq t u → FuncEq (.lam t) (.lam u)
  | eta {n} {Γ : Ctx n} (f : Func Γ) :
      FuncEq (.lam (.app ((Ren.weaken Γ).funcMap f) (.var 0))) f
  | sub {n m} {Γ : Ctx n} {Δ : Ctx m} (σ : Sub Δ Γ) (hs : σ.Typed)
      {f g : Func Γ} : FuncEq f g → FuncEq (σ.funcSub f) (σ.funcSub g)
end

theorem eta_type {n} {Γ : Ctx n} (f : Func Γ) :
    (Dep.app ((Ren.weaken Γ).funcMap f) (.var 0)).type = f.type := by
  change instantiate ((Ren.weaken Γ).funcMap f).type (.var 0) = f.type
  rw [Ren.funcMap_type _ (Ren.typed_weaken Γ)]
  simp only [instantiate, Base.sub_comp]
  have h : (fun i => (up (Ren.weaken Γ).base i).sub (Fin.cases (Base.var 0) Base.var)) = Base.var := by
    funext i
    refine Fin.cases rfl (fun j => ?_) i
    rfl
  rw [h, Base.sub_id]

mutual
 theorem DepEq.type : {n : Nat} → {Γ : Ctx n} → {t u : Dep Γ} → DepEq t u → t.type = u.type
   | _, _, _, _, .refl _ => rfl
   | _, _, _, _, .symm h => (DepEq.type h).symm
   | _, _, _, _, .trans h k => (DepEq.type h).trans (DepEq.type k)
   | _, _, _, _, .app h a => congrArg (fun b => instantiate b a) (FuncEq.type h)
   | _, Γ, _, _, .beta t a => (Sub.depSub_type _ (Sub.typed_plug Γ a) t).symm
   | _, _, _, _, .sub σ hs h => (Sub.depSub_type σ hs _).trans
       ((congrArg (Base.sub σ.base) (DepEq.type h)).trans (Sub.depSub_type σ hs _).symm)
 theorem FuncEq.type : {n : Nat} → {Γ : Ctx n} → {f g : Func Γ} → FuncEq f g → f.type = g.type
   | _, _, _, _, .refl _ => rfl
   | _, _, _, _, .symm h => (FuncEq.type h).symm
   | _, _, _, _, .trans h k => (FuncEq.type h).trans (FuncEq.type k)
   | _, _, _, _, .lam h => DepEq.type h
   | _, _, _, _, .eta f => eta_type f
   | _, _, _, _, .sub σ hs h => (Sub.funcSub_type σ hs _).trans
       ((congrArg (Base.sub (up σ.base)) (FuncEq.type h)).trans (Sub.funcSub_type σ hs _).symm)
end

/-- Beta after an arbitrary typed substitution, including a nonidentity base
assignment. This is derived from congruence and beta, not a primitive rule. -/
theorem substituted_beta {n m} {Γ : Ctx n} {Δ : Ctx m} (σ : Sub Δ Γ) (hs : σ.Typed)
    (t : Dep (.base Γ)) (a : Base n) :
    DepEq (.app (.lam (σ.lift.depSub t)) (a.sub σ.base))
      (σ.depSub ((Sub.plug Γ a).depSub t)) :=
  .sub σ hs (.beta t a)

/-- A function parameter can be eta-expanded under substitution. -/
theorem substituted_eta {n m} {Γ : Ctx n} {Δ : Ctx m} (σ : Sub Δ Γ) (hs : σ.Typed)
    (f : Func Γ) :
    FuncEq (σ.funcSub (.lam (.app ((Ren.weaken Γ).funcMap f) (.var 0)))) (σ.funcSub f) :=
  .sub σ hs (.eta f)

end SquareKit.Internal.DependentFunctions
