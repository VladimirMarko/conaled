namespace GuardedWrongEndpoint

open Contextual TelescopePresentation

example (hx : TelescopePresentation.Tm (Ω := Ω) Γ A x)
    (hy : TelescopePresentation.Tm (Ω := Ω) Γ A y)
    (hz : TelescopePresentation.Tm (Ω := Ω) Γ A z)
    (hw : TelescopePresentation.Tm (Ω := Ω) Γ A w)

    (hq : TelescopePresentation.Tm (Ω := Ω) Γ (Contextual.identity A x z) q)
    (hr : TelescopePresentation.Tm (Ω := Ω) Γ (Contextual.identity A z w) r) : ThreePaths Ω Γ :=
  ⟨⟨A,x,y,p,hx,hy,hp⟩,z,w,q,r,hz,hw,hq,hr⟩
end GuardedWrongEndpoint
