-- These are executable observations, not new kernel equality theorems.
def category : CheckError → String
  | .invalid _ => "invalid"
  | .notImplemented _ => "notImplemented"
  | .internal _ => "internal"

def assertDiagnostic (label expectedCategory expectedText : String) (e : CheckError) : IO Unit := do
  unless category e == expectedCategory && toString e == expectedText do
    throw (IO.userError s!"{label}: unexpected {category e}: {e}")
  IO.println s!"CONTROL {label}: {category e}: {e}"

-- Only the error/position projection is proved; no diagnostic classification.
def position {α : Type} : Except (CheckError × Nat) α → Option Nat
  | .ok _ => none
  | .error (_, n) => some n
