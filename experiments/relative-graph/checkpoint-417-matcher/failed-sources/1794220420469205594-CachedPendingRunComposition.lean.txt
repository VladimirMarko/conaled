namespace CachedPendingRunComposition

set_option maxRecDepth 4000
set_option maxHeartbeats 1000000
def defnOps (ops : CoreFnsI) (sort : Nat → Expr → CheckCM Level) (cv : ConstantVal) (value : Expr) : CheckCM Unit := do
  flushC
  let ty ← ops.infer 0 cv.type
  let _ ← sort 0 ty
  let _ : Unit ← pure ()
  let vt ← ops.infer 0 value
  unless ← ops.defeq 0 vt cv.type do
    throw (.invalid s!"type mismatch in {ValueKind.defn.word} {cv.name}")
theorem defn_eq (fe : FEnv) (cv : ConstantVal) (value : Expr) (pos vis : Nat) :
    checkPending .verified fe ⟨⟨.defn,cv,value⟩,pos,vis⟩ =
      defnOps (coreKnotI .verified (fe.restrictTo vis) checkFuel)
        (opSIxC .verified (fe.restrictTo vis)) cv value := by
  rfl

theorem ops_run (ops : CoreFnsI) (sort : Nat → Expr → CheckCM Level)
    (cv : ConstantVal) (value ty vty : Expr) (u : Level) (s1 s2 s3 out : CState)
    (hi : ops.infer 0 cv.type {} = .ok (ty,s1))
    (hs : sort 0 ty s1 = .ok (u,s2))
    (hv : ops.infer 0 value s2 = .ok (vty,s3))
    (he : ops.defeq 0 vty cv.type s3 = .ok (true,out)) :
    defnOps ops sort cv value {} = .ok ((),out) := by
  unfold defnOps
  refine bind_ok _ _ {} {} out _ _ rfl ?_
  refine bind_ok _ _ {} s1 out _ _ hi ?_
  refine bind_ok _ _ s1 s2 out _ _ hs ?_
  refine bind_ok _ _ s2 s2 out _ _ rfl ?_
  refine bind_ok _ _ s2 s3 out _ _ hv ?_
  exact bind_ok _ _ s3 out out _ _ he rfl
end CachedPendingRunComposition
